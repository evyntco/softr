async function postFinalDetailsWebhook(payload) {
  if (!FINAL_DETAILS_WEBHOOK_URL) {
    return { skipped: true };
  }

  const response = await fetch(FINAL_DETAILS_WEBHOOK_URL, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(payload),
  });

  if (!response.ok) {
    throw new Error(
      `Webhook responded with ${response.status}`
    );
  }

  return { skipped: false };
}

  function getFinalDetailsPayload() {
    const includedPackages =
      formData.servicePackage
        .map((id) => servicePackageIdToLabel[id])
        .filter(Boolean)
        .map(servicePackageDisplayLabel);

    return {
      recordId: recordId || "",
      projectName: formData.projectName,
      eventType: formData.eventType,
      eventDate: formData.eventDate,

  async function confirmAndLock() {
    if (finalDetailsAlreadyConfirmed) {
      return;
    }

    if (previewMode) {
      toast.info(
        "Preview mode only. Open this page with a Beverage Configuration record to confirm and lock."
      );
      return;
    }

    if (!recordId || !updatesEnabled) {
      toast.error(
        "One or more Beverage Configuration write groups are not available."
      );
      return;
    }

    if (sessionIssues.length > 0) {
      toast.error(
        "Resolve the open items on Read Back before confirming and locking Final Details."
      );
      return;
    }

    setFinalDetailsBusy(true);

    try {
      const saved = await save({ confirming: true });

      if (!saved) {
        return;
      }

      try {
        const result = await postFinalDetailsWebhook(
          getFinalDetailsPayload()
        );

        if (!result.skipped) {
          toast.success(
            "Final details email sent."
          );
        }
      } catch (webhookError) {
        console.error(
          "[Final Details] Email webhook failed:",
          webhookError
        );

        toast.error(
          "Locked, but the confirmation email failed to send. Use Resend Final Details Email to try again."
        );
      }
    } finally {
      setFinalDetailsBusy(false);
    }
  }

  async function resendFinalDetailsEmail() {
    setFinalDetailsBusy(true);

    try {
      const result = await postFinalDetailsWebhook(
        getFinalDetailsPayload()
      );

      if (result.skipped) {
        toast.info(
          "No email webhook URL is configured yet."
        );
      } else {
        toast.success(
          "Final details email sent."
        );
      }
    } catch (error) {
      console.error(
        "[Final Details] Resend email failed:",
        error
      );

      toast.error(
        error?.message ||
        "Unable to send the final details email."
      );
    } finally {
      setFinalDetailsBusy(false);
    }
  }
