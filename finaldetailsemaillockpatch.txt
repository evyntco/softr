function valueToText(value) {
  if (value === null || value === undefined) return "";

  if (
    typeof value === "string" ||
    typeof value === "number"
  ) {
    return String(value);
  }

  if (typeof value === "boolean") {
    return value ? "Yes" : "No";
  }

  if (Array.isArray(value)) {
    return value
      .map((item) => valueToText(item))
      .filter(Boolean)
      .join(", ");
  }

  if (typeof value === "object") {
    if (value.label) return valueToText(value.label);
    if (value.name) return valueToText(value.name);
    if (value.title) return valueToText(value.title);
    if (value.value !== undefined) {
      return valueToText(value.value);
    }
  }

  return "";
}

function extractLinkedRecordId(value) {
  if (value === null || value === undefined) {
    return "";
  }

  if (Array.isArray(value)) {
    for (const item of value) {
      const found =
        extractLinkedRecordId(item);

      if (found) {
        return found;
      }
    }

    return "";
  }

  if (typeof value === "object") {
    const direct =
      value.id ||
      value.recordId ||
      value.record_id ||
      value._id;

    if (direct) {
      return String(direct);
    }

    if (
      value.value !== undefined &&
      value.value !== value
    ) {
      const nested =
        extractLinkedRecordId(
          value.value
        );

      if (nested) {
        return nested;
      }
    }
  }

  return "";
}

function arrayFromValue(value) {
  if (!value) return [];

  if (Array.isArray(value)) {
    return value
      .map((item) => valueToText(item).trim())
      .filter(Boolean);
  }

  return valueToText(value)
    .split(",")
    .map((item) => item.trim())
    .filter(Boolean);
}

function dateInputValue(value) {
  const raw = valueToText(value).trim();

  if (!raw) return "";

  const match = raw.match(
    /^(\d{4})-(\d{2})-(\d{2})/
  );

  if (match) {
    return `${match[1]}-${match[2]}-${match[3]}`;
  }

  const parsed = new Date(raw);

  if (Number.isNaN(parsed.getTime())) {
    return "";
  }

  return [
    parsed.getFullYear(),
    String(parsed.getMonth() + 1).padStart(2, "0"),
    String(parsed.getDate()).padStart(2, "0"),
  ].join("-");
}

function valueToBoolean(value) {
  if (value === true) return true;
  if (value === false) return false;

  const normalized = valueToText(value)
    .trim()
    .toLowerCase();

  return [
    "yes",
    "true",
    "1",
    "checked",
  ].includes(normalized);
}

function venueRuleBoolean(value) {
  if (value === true) return true;
  if (value === false) return false;

  const normalized =
    valueToText(value)
      .trim()
      .toLowerCase();

  if (!normalized) {
    return null;
  }

  const negativeSignals = [
    "no",
    "false",
    "not allowed",
    "not permitted",
    "prohibited",
    "not provided",
    "none",
    "n/a",
  ];

  if (
    negativeSignals.some(
      (signal) =>
        normalized === signal ||
        normalized.includes(signal)
    )
  ) {
    return false;
  }

  const positiveSignals = [
    "yes",
    "true",
    "allowed",
    "permitted",
    "provided",
    "available",
  ];

  if (
    positiveSignals.some(
      (signal) =>
        normalized === signal ||
        normalized.includes(signal)
    )
  ) {
    return true;
  }

  return null;
}

function numericValue(value) {
  const raw =
    valueToText(value).trim();

  if (!raw) {
    return null;
  }

  const cleaned =
    raw.replace(
      /[^0-9.-]/g,
      ""
    );

  if (!cleaned) {
    return null;
  }

  const parsed =
    Number(cleaned);

  return Number.isFinite(parsed)
    ? parsed
    : null;
}

function normalizeMatch(value) {
  return valueToText(value)
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, " ")
    .trim();
}

function timeInputValue(value) {
  const raw = valueToText(value).trim();
  if (!raw) return "";

  const direct = raw.match(/^(\d{1,2}):(\d{2})/);
  if (direct) {
    return `${String(Number(direct[1])).padStart(2, "0")}:${direct[2]}`;
  }

  const ampm = raw.match(
    /^(\d{1,2}):(\d{2})\s*(AM|PM)$/i
  );

  if (ampm) {
    let hour = Number(ampm[1]);
    const minute = ampm[2];
    const suffix = ampm[3].toUpperCase();

    if (suffix === "PM" && hour < 12) hour += 12;
    if (suffix === "AM" && hour === 12) hour = 0;

    return `${String(hour).padStart(2, "0")}:${minute}`;
  }

  return "";
}

function timeToMinutes(value) {
  const normalized = timeInputValue(value);
  if (!normalized) return null;

  const [hours, minutes] =
    normalized.split(":").map(Number);

  if (
    Number.isNaN(hours) ||
    Number.isNaN(minutes)
  ) {
    return null;
  }

  return hours * 60 + minutes;
}

function minutesToTimeValue(totalMinutes) {
  if (
    totalMinutes === null ||
    totalMinutes === undefined ||
    Number.isNaN(Number(totalMinutes))
  ) {
    return "";
  }

  const normalized =
    ((Number(totalMinutes) % 1440) + 1440) %
    1440;

  const hours =
    Math.floor(normalized / 60);

  const minutes =
    normalized % 60;

  return `${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}`;
}

function timeOptionLabel(value) {
  if (!value) return "";

  const minutes =
    timeToMinutes(value);

  if (minutes === null) {
    return valueToText(value);
  }

  const hours24 =
    Math.floor(minutes / 60);

  const minute =
    minutes % 60;

  const suffix =
    hours24 >= 12
      ? "PM"
      : "AM";

  const hours12 =
    hours24 % 12 || 12;

  return `${hours12}:${String(minute).padStart(2, "0")} ${suffix}`;
}

function formatTime(value) {
  const normalized = timeInputValue(value);
  if (!normalized) return "Unknown";

  const [hours, minutes] =
    normalized.split(":").map(Number);

  const date = new Date();
  date.setHours(hours, minutes, 0, 0);

  return date.toLocaleTimeString(undefined, {
    hour: "numeric",
    minute: "2-digit",
  });
}

function formatCurrencyValue(value, fallback = "—") {
  const parsed = Number(value);

  if (!Number.isFinite(parsed)) {
    return fallback;
  }

  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    maximumFractionDigits: 2,
  }).format(parsed);
}

function googleMapsUrl(location) {
  const text = valueToText(location).trim();
  if (!text) return "";

  return `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(text)}`;
}

function numberOrNull(value) {
  if (
    value === "" ||
    value === null ||
    value === undefined
  ) {
    return null;
  }

  const parsed = Number(value);

  return Number.isNaN(parsed)
    ? null
    : parsed;
}

function getRecordIdFromUrl() {
  if (typeof window === "undefined") {
    return "";
  }

  const params =
    new URLSearchParams(
      window.location.search
    );

  return (
    params.get("recordId") ||
    params.get("beverageRecordId") ||
    ""
  ).trim();
}

function TimeSelect({
  value,
  onChange,
  placeholder = "Select time",
  disabled = false,
  stepMinutes = 15,
}) {
  const normalized =
    timeInputValue(value);

  const options =
    useMemo(() => {
      const list = [];

      for (
        let minutes = 0;
        minutes < 1440;
        minutes += stepMinutes
      ) {
        const optionValue =
          minutesToTimeValue(
            minutes
          );

        list.push({
          value: optionValue,
          label:
            timeOptionLabel(
              optionValue
            ),
        });
      }

      /*
       * Preserve a saved non-standard time such as 4:05 PM
       * even though the default picker uses 15-minute steps.
       */
      if (
        normalized &&
        !list.some(
          (item) =>
            item.value ===
            normalized
        )
      ) {
        list.push({
          value: normalized,
          label:
            `${timeOptionLabel(normalized)} (saved)`,
        });

        list.sort(
          (a, b) =>
            (
              timeToMinutes(
                a.value
              ) ?? 0
            ) -
            (
              timeToMinutes(
                b.value
              ) ?? 0
            )
        );
      }

      return list;
    }, [
      normalized,
      stepMinutes,
    ]);

  return (
    <div className="relative">
      <select
        value={normalized}
        disabled={disabled}
        onChange={(event) =>
          onChange(
            event.target.value
          )
        }
        className="h-11 w-full appearance-none rounded-xl border border-input bg-background px-3 pr-10 text-sm font-medium disabled:bg-[#F4F2ED] disabled:text-[#81786D]"
      >
        <option value="">
          {placeholder}
        </option>

        {options.map(
          (option) => (
            <option
              key={option.value}
              value={option.value}
            >
              {option.label}
            </option>
          )
        )}
      </select>

      <ChevronDown className="pointer-events-none absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 text-[#958C80]" />
    </div>
  );
}

  function setPackageEnabled(
    optionId,
    enabled
  ) {
    const bartenderServiceId =
      servicePackageLabelToId["bartender service"];
    const otherAddOnsId =
      servicePackageLabelToId["other add-ons"];

    if (enabled && optionId !== bartenderServiceId) {
      setBartenderOnlyMode(false);
    }

    setFormData((current) => {
      const currentlyIncluded =
        current.servicePackage.includes(optionId);

      let nextPackages = current.servicePackage;

      if (enabled && !currentlyIncluded) {
        nextPackages = [...current.servicePackage, optionId];
      }

      if (!enabled && currentlyIncluded) {
        nextPackages = current.servicePackage.filter(
          (id) => id !== optionId
        );
      }

      return {
        ...current,
        servicePackage: nextPackages,
        ...(optionId === otherAddOnsId
          ? { isOtherAddOns: enabled }
          : {}),
      };
    });

    setDirty(true);
  }

  function toggleArrayField(
    field,
    option

    setFormData((current) => {
      const values =
        Array.isArray(
          current[field]

          ? current[field]
          : [];

        [field]:
          values.includes(option)
            ? values.filter(
              (item) =>
                item !== option

            : [
              ...values,
              option,

    });

    setDirty(true);
  }

        "One or more Beverage Configuration write groups are not available."

      eventHostName:
        formData.eventHostName.trim(),

      contactEmail:
        formData.contactEmail.trim(),

      contactPhone:
        formData.contactPhone.trim(),

      secondaryContactName:
        formData.secondaryContactName.trim(),

      secondaryContactPhone:
        formData.secondaryContactPhone.trim(),

      eventStatus:
        resolvedEventStatusValue,

      payment:
        formData.payment,

      totalPackageAmount:
        numberOrNull(
          formData.totalPackageAmount

      refundedAmount:
        numberOrNull(
          formData.refundedAmount
        ),

      accountAssignedTo:
        formData.accountAssignedTo.trim(),

      finalDetailsAssigned:
        formData.finalDetailsAssigned.trim(),

      servicePackage:

      bartenderCount:

      arrivalSetupTime:

  }

  const currentUserLabel =

  const finalDetailsAlreadyConfirmed =
    !previewMode &&
    valueToBoolean(
      eventRecord?.fields?.finalDetailsConfirmed
    );

    updateCoreEvent.status === "pending" ||
    updatePackageDetails.status === "pending" ||
    updateAddOnDetails.status === "pending";

  const venueLookupError =
    selectedVenueError ||

  const venueLookupLoading =
    selectedVenueLoading ||
    venueFetching;

  const dirtySodaBarId =
    servicePackageLabelToId["dirty soda bar"];
  const sodaJuiceId =

  const otherAddOnsId =

  const bartenderServiceIdForCheck =
    servicePackageLabelToId["bartender service"];

    !!otherAddOnsId &&

    !!alcoholProcurementId &&

  const hasPackagePlanSelection =

    hasOtherAddOns ||
    hasAlcoholProcurement;

  /*
   * FULL PACKAGE PO WORKFLOW
   *

   *

   * that use the standard PO workflow.
   */
  const hasPackagePoPlanning =
    hasCocktailMixers ||
    hasDirtySodaBar ||
    hasSodaJuice ||
    hasOtherAddOns;

  /*
   * AUTO ARRIVAL CALCULATION
   *
   * Operational anchor = whichever happens first:
   * - Wedding ceremony (when applicable)
   * - Bar service open
   *
   * Setup allowance:
   * - Bartender Only: 60 minutes
   * - Cocktail Mixers and/or Dirty Soda Bar with >50 guests: 90 minutes
   * - Otherwise: 60 minutes
   * - Glassware adds another 30 minutes
   */
  const barOpenMinutesForArrival =
    timeToMinutes(
      formData.barServiceStartTime
    );

  const ceremonyMinutesForArrival =
    formData.eventType ===
      "Wedding" &&
      formData.hasCeremony
      ? timeToMinutes(
        formData.ceremonyTime
      )
      : null;

  const arrivalAnchorMinutes =
    [
      barOpenMinutesForArrival,
      ceremonyMinutesForArrival,
    ]
      .filter(
        (value) =>
          value !== null
      )
      .reduce(
        (earliest, value) =>
          earliest === null ||
            value < earliest
            ? value
            : earliest,
        null
      );

  const arrivalAnchorIsCeremony =
    ceremonyMinutesForArrival !== null &&
    arrivalAnchorMinutes ===
    ceremonyMinutesForArrival;

  const enhancedPackageSetup =
    !bartenderOnlyMode &&
    Number(
      formData.totalGuestCount
    ) > 50 &&
    (
      hasCocktailMixers ||
      hasDirtySodaBar
    );

  const baseArrivalLeadMinutes =
    enhancedPackageSetup
      ? 90
      : 60;

  const glasswareSetupMinutes =
    formData.glasswareProvidedByHost
      ? 30
      : 0;

  const totalArrivalLeadMinutes =
    baseArrivalLeadMinutes +
    glasswareSetupMinutes;

  const calculatedArrivalSetupTime =
    arrivalAnchorMinutes !== null
      ? minutesToTimeValue(
        arrivalAnchorMinutes -
        totalArrivalLeadMinutes
      )
      : "";

  const arrivalAnchorLabel =
    arrivalAnchorIsCeremony
      ? "ceremony"
      : barOpenMinutesForArrival !== null
        ? "bar open"
        : "";

  useEffect(() => {
    if (
      formData.arrivalSetupTime ===
      calculatedArrivalSetupTime
    ) {
      return;
    }

    setFormData(
      (current) => ({
        ...current,
        arrivalSetupTime:
          calculatedArrivalSetupTime,
      })
    );

    /*
     * After a real record has hydrated, treat a changed calculated
     * arrival as an event change that should be saved.
     */
    if (
      loadedRecordId &&
      !previewMode
    ) {
      setDirty(true);
    }
  }, [
    calculatedArrivalSetupTime,
    formData.arrivalSetupTime,
    loadedRecordId,
    previewMode,
  ]);

  /*
   * VENUE VS. PLAN SOURCE OF TRUTH
   *
   * Do not derive these from Beverage Configuration, search
   * text, or an inferred venue match. These are direct values
   * from the exact selected Venue List record.
   */
  const venueFieldsValue =
    matchedVenue?.fields ||
    {};

  const matchedVenueName =
    valueToText(
      venueFieldsValue.name
    );

  const matchedVenueAddress =
    valueToText(
      venueFieldsValue.address
    );

  const venueShotsAllowed =
    matchedVenue
      ? venueRuleBoolean(
        venueFieldsValue.shotsAllowed
      )
      : null;

  const venueAlcoholAllowed =
    matchedVenue
      ? venueRuleBoolean(
        venueFieldsValue.alcoholAllowed
      )
      : null;

  const venueHasDesignatedBar =
    matchedVenue
      ? venueRuleBoolean(
        venueFieldsValue.barSetupProvided
      )
      : null;

  const venueHasIceMachine =
    matchedVenue
      ? venueRuleBoolean(
        venueFieldsValue.iceMachine
      )
      : null;

  const venueTravelFeeMinimum =
    matchedVenue
      ? (
        numericValue(
          venueFieldsValue.travelFee
        ) ?? 0
      )

  const venueServeTimeLimitHours =
    matchedVenue
      ? numericValue(

  const iceMachineGuidance =

  const venueMinimumBartenders =

  const orderReviewItems =

            "Staff count · included in contracted package",

            "Recorded add-on amount",

          basis:
            "Recorded add-on budget",
          amount:

              formData.wineQty

            "Recorded procurement budgets",

            alcoholProcurementBudgetTotal > 0

          item: "Travel",

            `${totalServiceStaff} staff`,
          basis:
            `${formatCurrencyValue(

            travelFeeTotal,

      formData.prepaidGratuity &&

          quantity:
            `${totalServiceStaff} staff`,

            `${formatCurrencyValue(

  const travelFeeBelowVenueMinimum =
    matchedVenue &&
    venueTravelFeeMinimum > 0 &&

  const selectedMixerRecipes =

  const baseStaffingRatio =

  const venueMinimumForRecommendation =
    Number.isFinite(

      venueMinimumBartenders > 0

      : 0;

    guestBasedRecommendedBartenders;

    recommendedBartenders > 0 &&
    bartenderStaffCount <

    );

    closeMinutes !== null

      closeMinutes;

      adjustedClose <
      startMinutes

      adjustedClose += 1440;
    }

    serviceDurationMinutes =
      adjustedClose -
      startMinutes;

  const serviceDurationHours =
    serviceDurationMinutes !== null
      ? serviceDurationMinutes /
      60
      : null;

  const exceedsVenueServeTime =
    venueServeTimeLimitHours !== null &&
    venueServeTimeLimitHours > 0 &&
    serviceDurationHours !== null &&
    serviceDurationHours >
    venueServeTimeLimitHours;

    serviceDurationHours !== null &&
    serviceDurationHours > 6;

  if (
    closeMinutes !== null &&
    outMinutesRaw !== null
  ) {
    let adjustedOut =
      outMinutesRaw;

    if (
      adjustedOut <
      closeMinutes
    ) {
      adjustedOut += 1440;
    }

    cleanupBufferMinutes =
      adjustedOut -
      closeMinutes;
  }

  const sessionIssues =
    useMemo(() => {

      if (!formData.eventType) {
        issues.push(
          "Confirm the event type."
        );

      if (!formData.eventDate) {
        issues.push(
          "Confirm the event date."
        );
      }

      if (
        formData.eventType ===
        "Wedding" &&
        formData.hasCeremony &&
        !formData.ceremonyTime
      ) {
        issues.push(
          "Wedding ceremony is marked Yes but Ceremony Time is not confirmed."
        );

      if (
        !formData.barServiceStartTime ||
        !formData.barServiceCloseTime
      ) {
        issues.push(
          "Confirm bar open and close times in Event Snapshot."
        );

      if (
        !formData.eventLocation
      ) {
        issues.push(
          "Confirm the event location."
        );
      }

      if (
        formData.eventLocation &&
        !matchedVenue
      ) {
        issues.push(
          "Confirm whether this is an unknown/private venue or match it to Venue List."
        );
      }

      if (
        !formData.arrivalSetupTime ||
        !formData.cleanUpOutTime
      ) {
        issues.push(
          "Complete the arrival and cleanup/out timeline."
        );
      }

      if (
        setupBufferMinutes !== null &&
        setupBufferMinutes < 60
      ) {
        issues.push(
          "Setup window is under the normal 60-minute planning allowance."
        );
      }

      if (
        cleanupBufferMinutes !== null &&
        cleanupBufferMinutes < 30
      ) {
        issues.push(
          "Cleanup/out window is under the default 30-minute planning allowance."
        );
      }

      if (
        !formData.totalGuestCount
      ) {
        issues.push(
          "Confirm total guest count."
        );
      }

      if (
        !formData.guestsConsumingAlcohol
      ) {
        issues.push(
          "Confirm the 21+ alcohol-consuming guest count."
        );
      }

      if (
        bartenderCountBelowPractice
      ) {
        issues.push(
          `Bartender staffing is below the guest-based recommendation of ${recommendedBartenders} bartender${recommendedBartenders === 1 ? "" : "s"}.`

      }

      if (
        venueStaffingBelowMinimum
      ) {
        issues.push(
          `Venue requires at least ${venueMinimumForRecommendation} total bar-service staff. Current plan has ${totalServiceStaff}; bartenders plus Dirty Soda attendants/additional servers may count toward this minimum.`
        );
      }

      if (
        matchedVenue &&
        venueAlcoholAllowed === false &&
        Number(
          formData.guestsConsumingAlcohol
        ) > 0
      ) {
        issues.push(
          "Venue List shows alcohol is not allowed, but this event currently has alcohol-consuming guests."
        );
      }

      if (
        matchedVenue &&
        venueShotsAllowed === false &&
        /\bshots?\b/i.test(
          formData.bartenderWillServe ||
          ""
        )
      ) {
        issues.push(
          "Venue List shows shots are not allowed, but the bartender service scope references shots."
        );
      }

      if (exceedsVenueServeTime) {

          "Confirm the number of bars."

          "Venue List shows no designated bar/setup. Confirm the host will provide a front bar table and a back service table, both with linens."

      if (
        matchedVenue &&
        venueHasIceMachine === null
      ) {
        issues.push(
          "Venue ice-machine availability is not confirmed. Verify ice availability and responsibility."
        );
      }

      if (
        hasCocktailMixers &&
        formData.mixerSelection.length === 0
      ) {
        issues.push(
          "Cocktail Mixers is included but no mixer/cocktail selections are confirmed."
        );
      }

      if (
        hasDirtySodaBar &&
        formData.sodaSelections.length === 0
      ) {
        issues.push(
          "Dirty Soda Bar is included but no soda selections are confirmed."
        );

        hasSodaJuice &&
        formData.additionalSodaSelections.length === 0 &&
        formData.additionalJuiceSelections.length === 0
      ) {

          "Soda/Juice is included but no beverage selections are confirmed."
        );

        hasAlcoholProcurement &&
        !formData.liquorQty &&
        !formData.wineQty &&
        !formData.beerSeltzerQty
      ) {
        issues.push(
          "Alcohol Procurement is included but alcohol quantities are not confirmed."
        );

      if (
        !formData.bartenderWillServe
      ) {
        issues.push(
          "Document exactly what the bartender will serve."
        );

      if (
        !formData.finalDetailsAssigned.trim()
      ) {

          "Final check: identify who completed the Final Details Session."
        );

    }, [

      bartenderCountBelowPractice,
      venueStaffingBelowMinimum,

  useEffect(() => {

  const stepCompletion = {

        formData.eventDate &&
        formData.eventLocation &&
        formData.eventHostName &&
        formData.contactEmail &&
        formData.barServiceStartTime &&

    sessionSteps.findIndex(

    sessionSteps[
    activeStepIndex >= 0

  const guideByStep = {

      ask:
        "Start with the outcome, then lock the timing: “Walk me through the event and what you want the bar experience to feel like.” For weddings, confirm whether there is a ceremony and when it starts; then confirm exactly when the bar opens and closes.",
      practice:
        "Use the primary host for planning decisions and always identify a separate day-of contact when possible. Venue selection, service-time validation, arrival calculation, and cleanup timing are all confirmed here before moving into guest and service planning.",
      watch:
        "If the venue or address is still vague, do not treat venue rules, access, ice, travel, or setup assumptions as confirmed.",

    guests: {
      ask:
        "Separate total attendance from drinking attendance: “How many total guests, how many are 21+, and how many do you realistically expect to drink?”",
      practice:

      watch:
        "Do not size mixers, cups, ice, or staffing from total attendance if a meaningful portion of guests will not be drinking.",

Water responsibility must always be explicit. A Cheers water dispenser is not available on Bartender Only events.",

      ask:
        "Finish by reading the event back in plain language: timing, guest counts, staffing, what Cheers provides, what the host provides, venue restrictions, and unresolved items.",

      Math.min(

      sessionSteps[nextIndex].id

        activeStepIndex - 1,

      sessionSteps[

      typeof window !==
      "undefined"

        top: 0,
        behavior: "smooth",

              <Badge
                variant="outline"
                className={
                  previewMode
                    ? "border-blue-200 bg-blue-50 text-blue-800"

                      ? "border-amber-300 bg-amber-50 text-amber-800"

                }

                {saving ? (

          {previewMode && (
            <div className="mb-4 rounded-2xl border border-blue-200 bg-blue-50 px-4 py-3">
              <div className="flex items-start gap-3">
                <CircleAlert className="mt-0.5 h-5 w-5 shrink-0 text-blue-700" />

            <div className="grid gap-7 p-6 sm:p-8 lg:grid-cols-[minmax(0,1fr)_420px] lg:p-9">
              <div>
                <div className="flex flex-wrap items-center gap-2">
                  <Badge className="bg-[#D6B678] text-[#171613] hover:bg-[#D6B678]">
                    Final Details Session

                  {matchedVenue && (
                    <Badge
                      variant="outline"
                      className="border-emerald-400/30 bg-emerald-400/10 text-emerald-100"
                    >
                      Venue guidance connected
                    </Badge>
                  )}

                  {bartenderOnlyMode && (
                    <Badge
                      variant="outline"
                      className="border-blue-400/30 bg-blue-400/10 text-blue-100"

                      Bartender Only
                    </Badge>

                </div>

                <h1 className="mt-4 text-3xl font-semibold tracking-[-0.035em] sm:text-4xl">

                    sessionIssues.length

                  }
                  tone={
                    sessionIssues.length

                  }

                  }

                    venueStaffingBelowMinimum

                      : bartenderCountBelowPractice
                        ? "Bartender count below guest-based guide"
                        : venueMinimumForRecommendation > 0
                          ? `Venue minimum: ${venueMinimumForRecommendation} total service staff`
                          : drinkingGuests
                            ? hasCocktailMixers

                              : "Beer/wine bartender baseline"
                            : "Add drinking guest count"
                  }
                  tone={
                    staffingBelowPractice

                      : recommendedBartenders ||
                        venueMinimumForRecommendation

                            placeholder="Planner name"
                            className="bg-white sm:max-w-md"
                          />

                          {currentUserLabel && (
                            <Button
                              type="button"
                              variant="outline"
                              className="bg-white"
                              onClick={() =>
                                change(
                                  "finalDetailsAssigned",
                                  currentUserLabel
                                )
                              }
                            >
                              <UserRound className="mr-2 h-4 w-4" />
                              Me
                            </Button>
                          )}
                        </div>
                      </div>

                      {formData.finalDetailsAssigned.trim() && (
                        <Badge
                          variant="outline"
                          className="border-emerald-200 bg-white text-emerald-800"
                        >
                          <Check className="mr-1 h-3.5 w-3.5" />
                          Completed by {formData.finalDetailsAssigned}
                        </Badge>
                      )}
                    </div>
                  </section>

                  <div className="flex flex-col gap-3 rounded-[24px] bg-[#191815] p-5 text-white shadow-xl sm:flex-row sm:items-center sm:justify-between">
                    <div>
                      <p className="font-semibold">
                        Finish the session
                      </p>

                      <p className="mt-1 text-sm text-white/55">
                        Confirm the Order Review, identify who completed Final Details, then save the handoff.
                      </p>
                    </div>

                    <Button
                      onClick={save}
                      disabled={
                        previewMode ||
                        saving ||
                        !dirty ||
                        !updatesEnabled
                      }
                      className="bg-[#D7B777] text-[#171612] hover:bg-[#E4C992]"
                    >
                      {saving ? (
                        <Loader2 className="mr-2 h-4 w-4 animate-spin" />
                      ) : (
                        <Save className="mr-2 h-4 w-4" />
                      )}

                      Save Final Details
                    </Button>
                  </div>
                </div>
              )}

              {/* BOTTOM STEP NAV */}
              <div className="mt-5 flex items-center justify-between rounded-2xl border border-[#E3DED5] bg-white p-3 shadow-sm">
                <Button
                  type="button"
                  variant="ghost"
                  onClick={goToPreviousStep}
                  disabled={
                    activeStepIndex === 0
                  }
                >
                  <ArrowLeft className="mr-2 h-4 w-4" />
                  Previous
                </Button>

                <span className="text-xs font-semibold text-[#8B8277]">
                  {activeStepIndex + 1} / {sessionSteps.length}
                </span>

                {activeStepIndex <
                  sessionSteps.length - 1 ? (
                  <Button
                    type="button"
                    onClick={goToNextStep}
                    className="bg-[#1A1916] text-white hover:bg-[#34312C]"
                  >
                    Next
                    <ArrowRight className="ml-2 h-4 w-4" />
                  </Button>
                ) : (
                  <Button
                    type="button"
                    onClick={save}
                    disabled={
                      previewMode ||
                      saving ||
                      !dirty ||
                      !updatesEnabled
                    }
                    className="bg-[#1A1916] text-white hover:bg-[#34312C]"
                  >
                    Save
                  </Button>
                )}
              </div>
            </main>

            {/* STICKY CONVERSATION GUIDE */}
            <aside className="lg:sticky lg:top-5 lg:self-start">
              <div className="space-y-4">
                <section className="rounded-[24px] bg-[#1B1A17] p-5 text-white shadow-lg">
                  <div className="flex items-center gap-2">
                    <MessageCircle className="h-5 w-5 text-[#D6B678]" />

                    <div>
                      <p className="text-[10px] font-bold uppercase tracking-[0.1em] text-[#D6B678]">
                        Conversation Guide
                      </p>

                      <p className="mt-0.5 font-semibold">
                        {activeStepMeta.title}
                      </p>
                    </div>
                  </div>

                  <div className="mt-5 space-y-4">
                    <div>
                      <p className="text-[10px] font-bold uppercase tracking-[0.08em] text-white/40">
                        Ask this now
                      </p>

                      <p className="mt-1 text-sm leading-6 text-white/75">
                        {currentGuide.ask}
                      </p>
                    </div>

                    <div className="border-t border-white/10 pt-4">
                      <p className="text-[10px] font-bold uppercase tracking-[0.08em] text-white/40">
                        Standard practice
                      </p>

                      <p className="mt-1 text-sm leading-6 text-white/75">
                        {currentGuide.practice}
                      </p>
                    </div>

                    <div className="border-t border-white/10 pt-4">
                      <p className="text-[10px] font-bold uppercase tracking-[0.08em] text-white/40">
                        Watch for
                      </p>

                      <p className="mt-1 text-sm leading-6 text-white/75">
                        {currentGuide.watch}
                      </p>
                    </div>
                  </div>
                </section>

                <section className="rounded-[24px] border border-[#E3DED5] bg-white p-5 shadow-sm">
                  <div className="flex items-center justify-between gap-2">
                    <div>
                      <p className="text-[10px] font-bold uppercase tracking-[0.09em] text-[#A07943]">
                        Live Session Notes
                      </p>

                      <p className="mt-1 text-sm font-semibold">
                        Keep notes while they talk
                      </p>
                    </div>

                    <Badge variant="outline">
                      Saves to event
                    </Badge>
                  </div>

                  <textarea
                    value={
                      formData.additionalServiceDetails
                    }
                    onChange={(event) =>
                      change(
                        "additionalServiceDetails",
                        event.target.value
                      )
                    }
                    rows={7}
                    placeholder="Client priorities, exceptions, follow-ups, venue concerns..."
                    className="mt-4 w-full resize-y rounded-xl border border-[#DED8CE] bg-[#FAF8F4] px-3 py-3 text-sm leading-6"
                  />
                </section>

                <section className="rounded-[24px] border border-[#E3DED5] bg-white p-5 shadow-sm">
                  <div className="flex items-center gap-2">
                    <ListChecks className="h-5 w-5 text-[#A07943]" />

                    <div>
                      <p className="font-semibold">
                        Still unresolved
                      </p>

                      <p className="text-xs text-[#81786D]">
                        Automatic planning checks
                      </p>
                    </div>
                  </div>

                  <div className="mt-4 space-y-2">
                    {sessionIssues.length ? (
                      sessionIssues
                        .slice(0, 6)
                        .map(
                          (
                            issue,
                            index
                          ) => (
                            <div
                              key={`${issue}-${index}`}
                              className="flex items-start gap-2 rounded-xl bg-[#FAF8F4] px-3 py-2.5"
                            >
                              <CircleAlert className="mt-0.5 h-4 w-4 shrink-0 text-amber-700" />

                              <p className="text-xs leading-5 text-[#6E665C]">
                                {issue}
                              </p>
                            </div>
                          )
                        )
                    ) : (
                      <div className="rounded-xl bg-emerald-50 px-3 py-3 text-sm text-emerald-800">
                        No automatic planning gaps detected.
                      </div>
                    )}

                    {sessionIssues.length >
                      6 && (
                        <button
                          type="button"
                          onClick={() =>
                            setActiveStep(
                              "review"
                            )
                          }
                          className="w-full rounded-xl border border-[#DED8CE] px-3 py-2 text-xs font-semibold"
                        >
                          Review all {sessionIssues.length} items
                        </button>
                      )}
                  </div>
                </section>
              </div>
            </aside>
          </div>

          {/* MOBILE/STICKY SAVE */}
          <div className="sticky bottom-3 z-40 mt-6 lg:hidden">
            <div className="flex items-center justify-between gap-3 rounded-2xl bg-[#191815] p-3 text-white shadow-2xl">
              <div className="min-w-0">
                <p className="truncate text-sm font-semibold">
                  {dirty
                    ? "Unsaved session changes"
                    : "Session saved"}
                </p>

                <p className="text-[10px] text-white/45">
                  {sessionIssues.length} open item
                  {sessionIssues.length === 1
                    ? ""
                    : "s"}
                </p>
              </div>

              <Button
                onClick={save}
                disabled={
                  previewMode ||
                  saving ||
                  !dirty ||
                  !updatesEnabled
                }
                className="shrink-0 bg-[#D7B777] text-[#171612] hover:bg-[#E4C992]"
              >
                <Save className="mr-2 h-4 w-4" />
                Save
              </Button>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
